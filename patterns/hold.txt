#Pattern - 10: Half Diamond Star Pattern
class solution:
    def normal(self,n):
        for i in range(n):
            for j in range(i+1):
                print("*",end="")
            print()
    def inverted(self,n):
        for i in range(n):
            for j in range(n-i):
                print("*",end="")
            print()

if __name__=="__main__":
    sol=solution()
    n=int(input("enter number:"))
    sol.normal(n)
    sol.inverted(n)
